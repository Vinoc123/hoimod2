# Tunable peace numbers. Building bonuses live in common/buildings/hoimod2_buildings.txt.
# Population loss is 0.4% per month, floored at 50,000, in on_monthly.
# Leverage = 0.70 * enemy surrender progress + 0.30 * our share of the pair's casualties.
# Casualty share uses thousands and adds 80 before dividing.
# Advantage: war longer than 90 days, best leverage at least 0.35, gap at least 0.20.
# Frozen front and no shared front: longer than 365 days.
# The peace opens a real conference. Both countries sit as one losing alliance.
# Each player is moved to an off-map bench country. Those two benches divide only the
# disputed states: land a side still holds that it already owned stays off the table.
# Score is 1/1.35 of the disputed land, so the table can buy that land at full price.
# Afterwards each bench's states return to its original country.

hoimod_monthly_update = {
	if = {
		limit = { has_war = yes }
		set_variable = { hoimod_sp_6 = hoimod_sp_5 }
		set_variable = { hoimod_sp_5 = hoimod_sp_4 }
		set_variable = { hoimod_sp_4 = hoimod_sp_3 }
		set_variable = { hoimod_sp_3 = hoimod_sp_2 }
		set_variable = { hoimod_sp_2 = hoimod_sp_1 }
		set_variable = { hoimod_sp_1 = hoimod_sp_0 }
		set_variable = { hoimod_sp_0 = surrender_progress }
		add_to_variable = { hoimod_sp_samples = 1 }
		set_temp_variable = { rise = hoimod_sp_0 }
		subtract_from_temp_variable = { rise = hoimod_sp_6 }
		set_variable = { hoimod_sp_rise = rise }

		every_enemy_country = {
			limit = { is_subject = no }
			PREV = {
				set_variable = { hoimod_case@PREV = 0 }
				set_variable = { hoimod_ai_weight@PREV = 0 }
				set_variable = { hoimod_can_light@PREV = 0 }
				set_variable = { hoimod_can_medium@PREV = 0 }
				set_variable = { hoimod_can_heavy@PREV = 0 }
			}
			save_event_target_as = hoimod_pair_enemy
			PREV = { hoimod_update_pair = yes }
		}
	}
	else = {
		set_variable = { hoimod_sp_samples = 0 }
		set_variable = { hoimod_sp_rise = 0 }
	}
}

hoimod_update_pair = {
	set_variable = { hoimod_pair_enemy_var = event_target:hoimod_pair_enemy }

	set_temp_variable = { hoimod_len = global.num_days }
	subtract_from_temp_variable = { hoimod_len = hoimod_war_start@var:hoimod_pair_enemy_var }

	set_variable = { hoimod_pair_count = 0 }
	every_controlled_state = {
		limit = { is_owned_by = event_target:hoimod_pair_enemy }
		add_to_variable = { PREV.hoimod_pair_count = 1 }
	}
	set_temp_variable = { hoimod_i_hold = hoimod_pair_count }

	event_target:hoimod_pair_enemy = {
		set_variable = { hoimod_pair_count = 0 }
		every_controlled_state = {
			limit = { is_owned_by = ROOT }
			add_to_variable = { PREV.hoimod_pair_count = 1 }
		}
	}
	set_temp_variable = { hoimod_they_hold = event_target:hoimod_pair_enemy.hoimod_pair_count }

	set_temp_variable = { hoimod_count_ours = 1 }
	hoimod_measure_inflicted = yes
	set_temp_variable = { hoimod_us_on_them = hoimod_measured }

	set_temp_variable = { hoimod_count_ours = 0 }
	hoimod_measure_inflicted = yes
	set_temp_variable = { hoimod_them_on_us = hoimod_measured }

	set_temp_variable = { denom = hoimod_us_on_them }
	add_to_temp_variable = { denom = hoimod_them_on_us }
	add_to_temp_variable = { denom = 80 }
	set_temp_variable = { our_share = hoimod_us_on_them }
	divide_temp_variable = { our_share = denom }
	set_temp_variable = { their_share = hoimod_them_on_us }
	divide_temp_variable = { their_share = denom }

	set_temp_variable = { lev_us = event_target:hoimod_pair_enemy.surrender_progress }
	multiply_temp_variable = { lev_us = 0.70 }
	set_temp_variable = { share_part = our_share }
	multiply_temp_variable = { share_part = 0.30 }
	add_to_temp_variable = { lev_us = share_part }

	set_temp_variable = { lev_them = surrender_progress }
	multiply_temp_variable = { lev_them = 0.70 }
	set_temp_variable = { share_part = their_share }
	multiply_temp_variable = { share_part = 0.30 }
	add_to_temp_variable = { lev_them = share_part }

	set_temp_variable = { signed = lev_us }
	subtract_from_temp_variable = { signed = lev_them }
	set_temp_variable = { gap = signed }
	if = {
		limit = { check_variable = { gap < 0 } }
		multiply_temp_variable = { gap = -1 }
	}

	set_temp_variable = { cas_sum = hoimod_us_on_them }
	add_to_temp_variable = { cas_sum = hoimod_them_on_us }

	set_temp_variable = { case = 0 }
	set_temp_variable = { base_score = 0 }
	if = {
		limit = {
			check_variable = { hoimod_len > 89 }
			OR = {
				check_variable = { lev_us > 0.349 }
				check_variable = { lev_them > 0.349 }
			}
			check_variable = { gap > 0.199 }
		}
		set_temp_variable = { case = 1 }
		set_temp_variable = { base_score = 0 }
	}
	else_if = {
		limit = {
			check_variable = { hoimod_len > 364 }
			check_variable = { hoimod_sp_samples > 6 }
			check_variable = { event_target:hoimod_pair_enemy.hoimod_sp_samples > 6 }
			surrender_progress > 0.049
			NOT = { surrender_progress > 0.55 }
			event_target:hoimod_pair_enemy = {
				surrender_progress > 0.049
				NOT = { surrender_progress > 0.55 }
			}
			check_variable = { hoimod_sp_rise < 0.081 }
			check_variable = { event_target:hoimod_pair_enemy.hoimod_sp_rise < 0.081 }
			check_variable = { cas_sum > 299 }
		}
		set_temp_variable = { case = 2 }
		set_temp_variable = { base_score = 0 }
	}
	else_if = {
		limit = {
			check_variable = { hoimod_len > 364 }
			NOT = {
				any_controlled_state = {
					any_neighbor_state = {
						is_controlled_by = event_target:hoimod_pair_enemy
					}
				}
			}
			check_variable = { hoimod_i_hold < 2 }
			check_variable = { hoimod_they_hold < 2 }
		}
		set_temp_variable = { case = 3 }
		set_temp_variable = { base_score = 0 }
	}

	set_temp_variable = { adv_left = 90 }
	subtract_from_temp_variable = { adv_left = hoimod_len }
	if = {
		limit = { check_variable = { adv_left < 0 } }
		set_temp_variable = { adv_left = 0 }
	}
	set_temp_variable = { year_left = 365 }
	subtract_from_temp_variable = { year_left = hoimod_len }
	if = {
		limit = { check_variable = { year_left < 0 } }
		set_temp_variable = { year_left = 0 }
	}

	set_variable = { hoimod_case@var:hoimod_pair_enemy_var = case }
	set_variable = { hoimod_base_score@var:hoimod_pair_enemy_var = base_score }
	set_variable = { hoimod_lev@var:hoimod_pair_enemy_var = lev_us }
	set_variable = { hoimod_their_lev@var:hoimod_pair_enemy_var = lev_them }
	set_variable = { hoimod_gap@var:hoimod_pair_enemy_var = gap }
	set_variable = { hoimod_signed_gap@var:hoimod_pair_enemy_var = signed }
	set_variable = { hoimod_us_on_them@var:hoimod_pair_enemy_var = hoimod_us_on_them }
	set_variable = { hoimod_them_on_us@var:hoimod_pair_enemy_var = hoimod_them_on_us }
	set_variable = { hoimod_held@var:hoimod_pair_enemy_var = hoimod_i_hold }
	set_variable = { hoimod_they_held@var:hoimod_pair_enemy_var = hoimod_they_hold }
	set_variable = { hoimod_adv_left@var:hoimod_pair_enemy_var = adv_left }
	set_variable = { hoimod_year_left@var:hoimod_pair_enemy_var = year_left }
	set_variable = { hoimod_can_light@var:hoimod_pair_enemy_var = 0 }
	set_variable = { hoimod_can_medium@var:hoimod_pair_enemy_var = 0 }
	set_variable = { hoimod_can_heavy@var:hoimod_pair_enemy_var = 0 }
	if = {
		limit = { check_variable = { case > 0 } }
		set_variable = { hoimod_can_light@var:hoimod_pair_enemy_var = 1 }
	}
	if = {
		limit = { check_variable = { case = 1 } }
		set_variable = { hoimod_can_medium@var:hoimod_pair_enemy_var = 1 }
		set_variable = { hoimod_can_heavy@var:hoimod_pair_enemy_var = 1 }
	}

	set_temp_variable = { aiw = 0 }
	if = {
		limit = {
			check_variable = { case = 1 }
			check_variable = { signed > 0.199 }
		}
		set_temp_variable = { aiw = gap }
		subtract_from_temp_variable = { aiw = 0.20 }
		divide_temp_variable = { aiw = 0.40 }
		multiply_temp_variable = { aiw = 25 }
		add_to_temp_variable = { aiw = 70 }
		clamp_temp_variable = { var = aiw min = 70 max = 95 }
	}
	else_if = {
		limit = {
			check_variable = { case = 1 }
			check_variable = { signed < -0.199 }
		}
		set_temp_variable = { aiw = 25 }
		if = {
			limit = { surrender_progress > 0.399 }
			set_temp_variable = { aiw = 60 }
		}
	}
	else_if = {
		limit = { check_variable = { case = 2 } }
		set_temp_variable = { aiw = 55 }
		if = {
			limit = { check_variable = { hoimod_them_on_us > hoimod_us_on_them } }
			add_to_temp_variable = { aiw = 20 }
		}
		set_temp_variable = { our_ic = num_of_factories }
		set_temp_variable = { their_ic = event_target:hoimod_pair_enemy.num_of_factories }
		set_temp_variable = { their_sp = event_target:hoimod_pair_enemy.surrender_progress }
		if = {
			limit = {
				check_variable = { our_ic > their_ic }
				check_variable = { surrender_progress < their_sp }
			}
			add_to_temp_variable = { aiw = -30 }
		}
	}
	else_if = {
		limit = { check_variable = { case = 3 } }
		set_temp_variable = { aiw = 40 }
		if = {
			limit = { has_government = democratic }
			add_to_temp_variable = { aiw = 20 }
		}
		if = {
			limit = {
				any_enemy_country = {
					NOT = { tag = event_target:hoimod_pair_enemy }
				}
			}
			add_to_temp_variable = { aiw = 20 }
		}
		if = {
			limit = {
				has_government = fascism
				is_major = yes
				has_wargoal_against = event_target:hoimod_pair_enemy
			}
			add_to_temp_variable = { aiw = -20 }
		}
	}
	if = {
		limit = { check_variable = { aiw < 0 } }
		set_temp_variable = { aiw = 0 }
	}
	set_variable = { hoimod_ai_weight@var:hoimod_pair_enemy_var = aiw }
}

hoimod_send_offer = {
	if = {
		limit = {
			NOT = { has_country_flag = hoimod_offer_pending }
			NOT = { is_in_peace_conference = yes }
		}
		FROM = { save_event_target_as = hoimod_pair_enemy }
		hoimod_update_pair = yes
		if = {
			limit = {
				check_variable = { hoimod_case@var:hoimod_pair_enemy_var > 0 }
				OR = {
					check_variable = { hoimod_tier = 0 }
					AND = {
						check_variable = { hoimod_tier = 1 }
						check_variable = { hoimod_can_light@var:hoimod_pair_enemy_var = 1 }
					}
					AND = {
						check_variable = { hoimod_tier = 2 }
						check_variable = { hoimod_can_medium@var:hoimod_pair_enemy_var = 1 }
					}
					AND = {
						check_variable = { hoimod_tier = 3 }
						check_variable = { hoimod_can_heavy@var:hoimod_pair_enemy_var = 1 }
					}
				}
			}
			set_variable = { global.hoimod_score = hoimod_base_score@var:hoimod_pair_enemy_var }
			set_variable = { global.hoimod_case = hoimod_case@var:hoimod_pair_enemy_var }
			set_variable = { global.hoimod_rep = hoimod_tier }
			set_variable = { global.hoimod_gap = hoimod_gap@var:hoimod_pair_enemy_var }
			set_variable = { global.hoimod_proposer = THIS }
			set_variable = { global.hoimod_target = event_target:hoimod_pair_enemy }

			set_temp_variable = { lev_us_now = hoimod_lev@var:hoimod_pair_enemy_var }
			set_temp_variable = { lev_them_now = hoimod_their_lev@var:hoimod_pair_enemy_var }
			if = {
				limit = { NOT = { check_variable = { lev_us_now < lev_them_now } } }
				set_variable = { global.hoimod_winner = THIS }
				set_variable = { global.hoimod_loser = event_target:hoimod_pair_enemy }
			}
			else = {
				set_variable = { global.hoimod_winner = event_target:hoimod_pair_enemy }
				set_variable = { global.hoimod_loser = THIS }
			}

			set_temp_variable = { accept = 30 }
			if = {
				limit = { check_variable = { global.hoimod_case = 1 } }
				if = {
					limit = { check_variable = { hoimod_signed_gap@var:hoimod_pair_enemy_var > 0 } }
					set_temp_variable = { accept = 30 }
					if = {
						limit = { event_target:hoimod_pair_enemy = { surrender_progress > 0.449 } }
						set_temp_variable = { accept = 80 }
					}
				}
				else = {
					set_temp_variable = { accept = global.hoimod_gap }
					subtract_from_temp_variable = { accept = 0.20 }
					divide_temp_variable = { accept = 0.40 }
					multiply_temp_variable = { accept = 25 }
					add_to_temp_variable = { accept = 70 }
					clamp_temp_variable = { var = accept min = 70 max = 95 }
				}
			}
			else_if = {
				limit = { check_variable = { global.hoimod_case = 2 } }
				set_temp_variable = { accept = 55 }
				if = {
					limit = { check_variable = { hoimod_us_on_them@var:hoimod_pair_enemy_var > hoimod_them_on_us@var:hoimod_pair_enemy_var } }
					add_to_temp_variable = { accept = 20 }
				}
				set_temp_variable = { our_ic = num_of_factories }
				set_temp_variable = { their_ic = event_target:hoimod_pair_enemy.num_of_factories }
				if = {
					limit = {
						check_variable = { their_ic > our_ic }
						event_target:hoimod_pair_enemy = { surrender_progress < ROOT.surrender_progress }
					}
					add_to_temp_variable = { accept = -30 }
				}
			}
			else = {
				set_temp_variable = { accept = 40 }
				if = {
					limit = { event_target:hoimod_pair_enemy = { has_government = democratic } }
					add_to_temp_variable = { accept = 20 }
				}
				if = {
					limit = {
						event_target:hoimod_pair_enemy = {
							any_enemy_country = {
								NOT = { tag = ROOT }
							}
						}
					}
					add_to_temp_variable = { accept = 20 }
				}
				if = {
					limit = {
						event_target:hoimod_pair_enemy = {
							has_government = fascism
							is_major = yes
							has_wargoal_against = ROOT
						}
					}
					add_to_temp_variable = { accept = -20 }
				}
			}
			clamp_temp_variable = { var = accept min = 1 max = 99 }
			set_variable = { global.hoimod_accept_weight = accept }
			set_temp_variable = { reject = 100 }
			subtract_from_temp_variable = { reject = accept }
			set_variable = { global.hoimod_reject_weight = reject }

			event_target:hoimod_pair_enemy = {
				if = {
					limit = { is_in_faction = yes }
					faction_leader = { save_event_target_as = hoimod_receiver }
				}
				else = {
					save_event_target_as = hoimod_receiver
				}
			}
			set_variable = { global.hoimod_receiver = event_target:hoimod_receiver }
			set_country_flag = hoimod_offer_pending
			event_target:hoimod_receiver = {
				set_country_flag = hoimod_offer_pending
				if = {
					limit = { is_ai = yes }
					country_event = { id = hoimod2.1 }
				}
				else = {
					set_country_flag = hoimod_player_must_answer
					var:global.hoimod_proposer = { save_global_event_target_as = hoimod_g_proposer }
					var:global.hoimod_target = { save_global_event_target_as = hoimod_g_target }
				}
			}
			event_target:hoimod_pair_enemy = { set_country_flag = hoimod_offer_pending }
		}
	}
}

hoimod_accept_offer = {

	hidden_effect = {
		set_variable = { global.hoimod_side_a = global.hoimod_proposer }
		set_variable = { global.hoimod_side_b = global.hoimod_target }
		hoimod_open_split = yes
	}

	load_focus_tree = generic_focus
	set_popularities = { neutrality = 100 }
	set_politics = { ruling_party = neutrality elections_allowed = no }
	create_country_leader = {
		name = "Delegate"
		expire = "1965.1.1"
		ideology = despotism
	}
}

hoimod_which_side = {
	set_variable = { global.hoimod_which_side = 0 }
	if = {
		limit = {
			OR = {
				tag = var:global.hoimod_side_a
				is_subject_of = var:global.hoimod_side_a
				is_in_faction_with = var:global.hoimod_side_a
				AND = {
					is_subject = yes
					overlord = { is_in_faction_with = var:global.hoimod_side_a }
				}
			}
		}
		set_variable = { global.hoimod_which_side = 1 }
	}
	else_if = {
		limit = {
			OR = {
				tag = var:global.hoimod_side_b
				is_subject_of = var:global.hoimod_side_b
				is_in_faction_with = var:global.hoimod_side_b
				AND = {
					is_subject = yes
					overlord = { is_in_faction_with = var:global.hoimod_side_b }
				}
			}
		}
		set_variable = { global.hoimod_which_side = 2 }
	}
}

hoimod_mark_states = {
	set_variable = { global.hoimod_pot = 0 }
	add_to_variable = { global.hoimod_mark_id = 1 }
	every_state = {
		limit = {
			owner = {
				OR = {
					tag = var:global.hoimod_side_a
					tag = var:global.hoimod_side_b
					is_subject_of = var:global.hoimod_side_a
					is_subject_of = var:global.hoimod_side_b
				}
			}
		}
		owner = { hoimod_which_side = yes }
		set_variable = { hoimod_owner_side = global.hoimod_which_side }
		controller = { hoimod_which_side = yes }
		set_variable = { hoimod_ctrl_side = global.hoimod_which_side }
		clear_variable = hoimod_claim
		if = {
			limit = {
				check_variable = { hoimod_owner_side = hoimod_ctrl_side }
				check_variable = { hoimod_ctrl_side > 0 }
			}
			owner = { set_variable = { PREV.hoimod_return_owner = THIS } }
			transfer_state_to = var:global.hoimod_vault
		}
		else = {
			set_variable = { hoimod_claim = hoimod_ctrl_side }
			set_variable = { hoimod_claim_mark = global.hoimod_mark_id }
			add_to_variable = { global.hoimod_pot = 1 }
		}
	}
}

hoimod_white_peace_pair = {
	clear_array = global.hoimod_wp_from
	clear_array = global.hoimod_wp_to
	set_variable = { global.hoimod_wp_i = 0 }
	every_country = {
		limit = {
			OR = {
				tag = var:global.hoimod_side_a
				is_subject_of = var:global.hoimod_side_a
				is_in_faction_with = var:global.hoimod_side_a
				AND = {
					is_subject = yes
					overlord = { is_in_faction_with = var:global.hoimod_side_a }
				}
			}
		}
		every_enemy_country = {
			limit = {
				OR = {
					tag = var:global.hoimod_side_b
					is_subject_of = var:global.hoimod_side_b
					is_in_faction_with = var:global.hoimod_side_b
					AND = {
						is_subject = yes
						overlord = { is_in_faction_with = var:global.hoimod_side_b }
					}
				}
			}
			add_to_array = { global.hoimod_wp_to = THIS }
			add_to_array = { global.hoimod_wp_from = PREV }
		}
	}
	while_loop_effect = {
		limit = { check_variable = { global.hoimod_wp_i < global.hoimod_wp_from^num } }
		set_temp_variable = { i = global.hoimod_wp_i }
		set_variable = { global.hoimod_wp_target = global.hoimod_wp_to^i }
		var:global.hoimod_wp_from^i = {
			white_peace = { tag = var:global.hoimod_wp_target }
		}
		add_to_variable = { global.hoimod_wp_i = 1 }
	}
}

hoimod_leave_for_split = {
	if = {
		limit = { is_in_faction = yes is_faction_leader = no }
		set_variable = { hoimod_old_leader = faction_leader }
		leave_faction = yes
	}
	else_if = {
		limit = {
			is_faction_leader = yes
			any_allied_country = { always = yes }
		}
		random_allied_country = { save_event_target_as = hoimod_keep_leader }
		set_variable = { hoimod_old_leader = event_target:hoimod_keep_leader }
		event_target:hoimod_keep_leader = { set_faction_leader = yes }
		leave_faction = yes
	}
	else_if = {
		limit = { is_faction_leader = yes }
		dismantle_faction = yes
	}
}

hoimod_open_split = {
	if = {
		limit = { check_variable = { global.hoimod_rep > 0 } }
		var:global.hoimod_winner = {
			set_country_flag = hoimod_rep_receiver
			set_variable = { hoimod_rep_tier = global.hoimod_rep }
			set_variable = { hoimod_rep_payer = global.hoimod_loser }
		}
		var:global.hoimod_loser = { set_country_flag = hoimod_rep_payer }
	}
	hoimod_clear_offer_flags = yes
	create_dynamic_country = {
		original_tag = KOM
		hoimod_init_bench = yes
		set_cosmetic_tag = HOIMOD_VAULT
		set_variable = { global.hoimod_vault = THIS }
	}
	create_dynamic_country = {
		original_tag = TAT
		hoimod_init_bench = yes
		set_cosmetic_tag = HOIMOD_BENCH_A
		set_country_flag = hoimod_conference_active
		set_variable = { global.hoimod_bench_a = THIS }
	}
	create_dynamic_country = {
		original_tag = BUK
		hoimod_init_bench = yes
		set_cosmetic_tag = HOIMOD_BENCH_B
		set_country_flag = hoimod_conference_active
		set_variable = { global.hoimod_bench_b = THIS }
	}
	hoimod_mark_states = yes
	if = {
		limit = { check_variable = { global.hoimod_pot = 0 } }
		var:global.hoimod_vault = {
			every_owned_state = { transfer_state_to = var:hoimod_return_owner }
			reserve_dynamic_country = no
		}
		var:global.hoimod_bench_a = { reserve_dynamic_country = no }
		var:global.hoimod_bench_b = { reserve_dynamic_country = no }
		hoimod_white_peace_pair = yes
		if = {
			limit = { var:global.hoimod_winner = { has_country_flag = hoimod_rep_receiver } }
			var:global.hoimod_winner = { hoimod_apply_reparations = yes }
		}
	}
	else = {
		hoimod_white_peace_pair = yes
		var:global.hoimod_side_a = { hoimod_leave_for_split = yes }
		var:global.hoimod_side_b = { hoimod_leave_for_split = yes }
		var:global.hoimod_side_a = {
			create_faction = hoimod_split_losers
			add_to_faction = var:global.hoimod_side_b
			set_country_flag = hoimod_temp_faction
		}
		var:global.hoimod_side_b = { set_country_flag = hoimod_temp_faction }
		var:global.hoimod_bench_a = {
			create_faction = hoimod_split_winners
			add_to_faction = var:global.hoimod_bench_b
		}
		if = {
			limit = { var:global.hoimod_side_a = { is_ai = no } }
			var:global.hoimod_bench_a = { change_tag_from = var:global.hoimod_side_a }
		}
		if = {
			limit = { var:global.hoimod_side_b = { is_ai = no } }
			var:global.hoimod_bench_b = { change_tag_from = var:global.hoimod_side_b }
		}
		var:global.hoimod_bench_a = {
			declare_war_on = { target = var:global.hoimod_side_a type = annex_everything }
			if = {
				limit = { NOT = { has_war_with = var:global.hoimod_side_b } }
				declare_war_on = { target = var:global.hoimod_side_b type = annex_everything }
			}
		}
		var:global.hoimod_bench_b = {
			add_to_war = {
				targeted_alliance = var:global.hoimod_bench_a
				enemy = var:global.hoimod_side_a
			}
		}
		every_state = {
			limit = {
				has_variable = hoimod_claim
				check_variable = { hoimod_claim_mark = global.hoimod_mark_id }
			}
			if = {
				limit = { check_variable = { hoimod_claim = 1 } }
				set_state_controller_to = var:global.hoimod_bench_a
			}
			else_if = {
				limit = { check_variable = { hoimod_claim = 2 } }
				set_state_controller_to = var:global.hoimod_bench_b
			}
		}
		set_global_flag = hoimod_split_pc
		var:global.hoimod_bench_a = {
			start_peace_conference = {
				tag = var:global.hoimod_side_a
				score_factor = 0.74
				winner_scope = FACTION
				loser_scope = FACTION
			}
		}
	}
}

hoimod_finish_split = {
	clr_global_flag = hoimod_split_pc
	if = {
		limit = {
			var:global.hoimod_bench_a = { is_ai = no }
			var:global.hoimod_side_a = { exists = yes }
		}
		var:global.hoimod_side_a = { change_tag_from = var:global.hoimod_bench_a }
	}
	if = {
		limit = {
			var:global.hoimod_bench_b = { is_ai = no }
			var:global.hoimod_side_b = { exists = yes }
		}
		var:global.hoimod_side_b = { change_tag_from = var:global.hoimod_bench_b }
	}
	var:global.hoimod_bench_a = {
		every_owned_state = { transfer_state_to = var:global.hoimod_side_a }
	}
	var:global.hoimod_bench_b = {
		every_owned_state = { transfer_state_to = var:global.hoimod_side_b }
	}
	var:global.hoimod_vault = {
		every_owned_state = { transfer_state_to = var:hoimod_return_owner }
		reserve_dynamic_country = no
	}
	var:global.hoimod_bench_a = {
		if = { limit = { is_faction_leader = yes } dismantle_faction = yes }
		reserve_dynamic_country = no
		clr_country_flag = hoimod_conference_active
	}
	var:global.hoimod_bench_b = {
		reserve_dynamic_country = no
		clr_country_flag = hoimod_conference_active
	}
	var:global.hoimod_side_a = {
		if = {
			limit = { is_faction_leader = yes has_country_flag = hoimod_temp_faction }
			dismantle_faction = yes
		}
		clr_country_flag = hoimod_temp_faction
	}
	var:global.hoimod_side_b = {
		if = {
			limit = { is_in_faction = yes has_country_flag = hoimod_temp_faction }
			leave_faction = yes
		}
		clr_country_flag = hoimod_temp_faction
	}
	var:global.hoimod_side_a = { hoimod_restore_faction = yes }
	var:global.hoimod_side_b = { hoimod_restore_faction = yes }
	if = {
		limit = { var:global.hoimod_winner = { has_country_flag = hoimod_rep_receiver } }
		var:global.hoimod_winner = { hoimod_apply_reparations = yes }
	}
}

hoimod_restore_faction = {
	if = {
		limit = {
			has_variable = hoimod_old_leader
			NOT = { is_in_faction = yes }
			var:hoimod_old_leader = { is_in_faction = yes }
		}
		var:hoimod_old_leader = { add_to_faction = PREV }
	}
	clear_variable = hoimod_old_leader
}

hoimod_reject_offer = {
	var:global.hoimod_proposer = {
		set_temp_variable = { until = global.num_days }
		add_to_temp_variable = { until = 180 }
		set_variable = { hoimod_cd_until@var:global.hoimod_target = until }
	}
	var:global.hoimod_target = {
		set_temp_variable = { until = global.num_days }
		add_to_temp_variable = { until = 180 }
		set_variable = { hoimod_cd_until@var:global.hoimod_proposer = until }
	}
	var:global.hoimod_receiver = {
		set_temp_variable = { until = global.num_days }
		add_to_temp_variable = { until = 180 }
		set_variable = { hoimod_cd_until@var:global.hoimod_proposer = until }
	}
	hoimod_clear_offer_flags = yes
}

hoimod_clear_offer_flags = {
	var:global.hoimod_proposer = { clr_country_flag = hoimod_offer_pending }
	var:global.hoimod_target = { clr_country_flag = hoimod_offer_pending }
	var:global.hoimod_receiver = {
		clr_country_flag = hoimod_offer_pending
		clr_country_flag = hoimod_player_must_answer
	}
}

hoimod_apply_reparations = {
	if = {
		limit = {
			has_country_flag = hoimod_rep_receiver
			check_variable = { hoimod_rep_tier > 0 }
		}
		var:hoimod_rep_payer = {
			if = {
				limit = {
					exists = yes
					any_owned_state = { always = yes }
				}
				if = {
					limit = { check_variable = { PREV.hoimod_rep_tier = 1 } }
					add_timed_idea = { idea = hoimod_rep_pay_light years = 1 }
					PREV = { add_timed_idea = { idea = hoimod_rep_get_light years = 1 } }
				}
				else_if = {
					limit = { check_variable = { PREV.hoimod_rep_tier = 2 } }
					add_timed_idea = { idea = hoimod_rep_pay_medium years = 2 }
					PREV = { add_timed_idea = { idea = hoimod_rep_get_medium years = 2 } }
				}
				else_if = {
					limit = { check_variable = { PREV.hoimod_rep_tier = 3 } }
					add_timed_idea = { idea = hoimod_rep_pay_heavy years = 3 }
					PREV = { add_timed_idea = { idea = hoimod_rep_get_heavy years = 3 } }
				}
			}
			clr_country_flag = hoimod_rep_payer
		}
	}
	clr_country_flag = hoimod_rep_receiver
	clr_country_flag = hoimod_conference_active
	clr_country_flag = hoimod_offer_pending
}
