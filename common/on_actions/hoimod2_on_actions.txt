on_actions = {

	on_startup = {
		effect = {
			every_country = {
				limit = { has_war = yes }
				hoimod_monthly_update = yes
			}
			every_country = {
				limit = {
					original_tag = GER
					has_government = fascism
				}
				set_technology = {
					hoimod_camp_construction = 1
					popup = no
				}
			}
		}
	}

	on_government_change = {
		effect = {
			if = {
				limit = {
					original_tag = GER
					has_government = fascism
					NOT = { has_tech = hoimod_camp_construction }
				}
				set_technology = {
					hoimod_camp_construction = 1
					popup = no
				}
			}
		}
	}

	on_state_control_changed = {
		effect = {
			FROM.FROM = {
				if = {
					limit = {
						hoimod_camp > 0
						OR = {
							is_owned_by = ROOT
							is_core_of = ROOT
							ROOT = {
								OR = {
									NOT = { original_tag = GER }
									NOT = { has_government = fascism }
								}
							}
						}
					}
					remove_building = {
						type = hoimod_camp
						level = 1
					}
				}
			}
		}
	}

	on_war_relation_added = {
		effect = {
			ROOT = {
				set_variable = { hoimod_war_start@FROM = global.num_days }
				set_variable = { hoimod_adv_left@FROM = 90 }
				set_variable = { hoimod_year_left@FROM = 365 }
			}
			FROM = {
				set_variable = { hoimod_war_start@ROOT = global.num_days }
				set_variable = { hoimod_adv_left@ROOT = 90 }
				set_variable = { hoimod_year_left@ROOT = 365 }
			}
		}
	}

	on_peaceconference_ended = {
		effect = {
			if = {
				limit = { has_global_flag = hoimod_split_pc }
				hoimod_finish_split = yes
			}
			if = {
				limit = { has_country_flag = hoimod_rep_receiver }
				hoimod_apply_reparations = yes
			}
			clr_country_flag = hoimod_conference_active
			clr_country_flag = hoimod_offer_pending
			every_country = {
				clr_country_flag = hoimod_conference_active
			}
		}
	}

	on_monthly = {
		effect = {
			hoimod_monthly_update = yes
		}
		effect = {
			if = {
				limit = {
					original_tag = GER
					has_government = fascism
					NOT = { has_tech = hoimod_camp_construction }
				}
				set_technology = {
					hoimod_camp_construction = 1
					popup = no
				}
			}
			if = {
				limit = {
					original_tag = GER
				}
				every_controlled_state = {
					limit = {
						hoimod_camp > 0
					}
					if = {
						limit = {
							OR = {
								is_owned_by = ROOT
								is_core_of = ROOT
								ROOT = { NOT = { has_government = fascism } }
							}
						}
						remove_building = {
							type = hoimod_camp
							level = 1
						}
					}
					else_if = {
						limit = {
							state_population > 50000
						}
						set_temp_variable = { hoimod_loss = state_population }
						multiply_temp_variable = { hoimod_loss = -0.004 }
						add_manpower = hoimod_loss
					}
				}
			}
		}
	}
}
